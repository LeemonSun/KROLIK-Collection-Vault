===== FILE: Source/KrolikBosses/HediffComp_KrolikRegeneration.cs =====
using System.Collections.Generic;
using System.Linq;
using Verse;

namespace KrolikBosses
{
    public class HediffCompProperties_KrolikRegeneration : HediffCompProperties
    {
        public int healIntervalTicks = 60;
        public float healAmount = 1f;
        public bool healAll = false;
        public bool regrowMissingParts = false;
        public int missingPartIntervalTicks = 600;

        public HediffCompProperties_KrolikRegeneration()
        {
            compClass = typeof(HediffComp_KrolikRegeneration);
        }
    }

    public class HediffComp_KrolikRegeneration : HediffComp
    {
        public HediffCompProperties_KrolikRegeneration Props =>
            (HediffCompProperties_KrolikRegeneration)props;

        public override void CompPostTickInterval(ref float severityAdjustment, int delta)
        {
            base.CompPostTickInterval(ref severityAdjustment, delta);

            Pawn pawn = parent.pawn;
            if (pawn == null || pawn.Dead || pawn.health == null)
                return;

            if (pawn.IsHashIntervalTick(Props.healIntervalTicks, delta))
                HealInjuries(pawn);

            if (Props.regrowMissingParts && pawn.IsHashIntervalTick(Props.missingPartIntervalTicks, delta))
                RestoreOneMissingPart(pawn);
        }

        private void HealInjuries(Pawn pawn)
        {
            List<Hediff_Injury> injuries = new List<Hediff_Injury>();
            List<Hediff> hediffs = pawn.health.hediffSet.hediffs;

            for (int i = 0; i < hediffs.Count; i++)
            {
                if (hediffs[i] is Hediff_Injury injury && injury.Severity > 0f)
                    injuries.Add(injury);
            }

            if (injuries.Count == 0)
                return;

            if (Props.healAll)
            {
                for (int i = 0; i < injuries.Count; i++)
                    injuries[i].Heal(Props.healAmount);
            }
            else
            {
                injuries.RandomElement().Heal(Props.healAmount);
            }
        }

        private void RestoreOneMissingPart(Pawn pawn)
        {
            Hediff_MissingPart missing = pawn.health.hediffSet.hediffs
                .OfType<Hediff_MissingPart>()
                .Where(x => x.Part != null)
                .Where(x => x.Part.parent == null || !pawn.health.hediffSet.PartIsMissing(x.Part.parent))
                .Where(x => !pawn.health.hediffSet.PartOrAnyAncestorHasDirectlyAddedParts(x.Part))
                .FirstOrDefault();

            if (missing?.Part == null)
                return;

            pawn.health.RestorePart(missing.Part);
        }
    }
}


===== FILE: Source/KrolikBosses/ThingComp_BossDamageController.cs =====
using RimWorld;
using Verse;

namespace KrolikBosses
{
    public class CompProperties_BossDamageController : CompProperties
    {
        public float maxDamagePerHit = 12f;
        public float phaseTwoHealthPercent = 0.55f;

        public CompProperties_BossDamageController()
        {
            compClass = typeof(ThingComp_BossDamageController);
        }
    }

    public class ThingComp_BossDamageController : ThingComp
    {
        private bool phaseTwoTriggered;

        private CompProperties_BossDamageController Props =>
            (CompProperties_BossDamageController)props;

        private Pawn Pawn => parent as Pawn;

        private static HediffDef BossCore => DefDatabase<HediffDef>.GetNamedSilentFail("KRL_BossCore");
        private static HediffDef PhaseTwo => DefDatabase<HediffDef>.GetNamedSilentFail("KRL_BossPhaseTwo");

        private bool IsBoss
        {
            get
            {
                Pawn pawn = Pawn;
                HediffDef core = BossCore;
                return pawn != null && core != null && pawn.health?.hediffSet?.HasHediff(core) == true;
            }
        }

        public override void PostExposeData()
        {
            base.PostExposeData();
            Scribe_Values.Look(ref phaseTwoTriggered, "phaseTwoTriggered", false);
        }

        public override void PostPreApplyDamage(ref DamageInfo dinfo, out bool absorbed)
        {
            base.PostPreApplyDamage(ref dinfo, out absorbed);

            if (!IsBoss || absorbed || !dinfo.Def.ExternalViolenceFor(parent))
                return;

            if (dinfo.Amount > Props.maxDamagePerHit)
                dinfo.SetAmount(Props.maxDamagePerHit);
        }

        public override void CompTickRare()
        {
            base.CompTickRare();

            Pawn pawn = Pawn;
            if (pawn == null || pawn.Dead || !IsBoss || phaseTwoTriggered)
                return;

            if (pawn.health.summaryHealth.SummaryHealthPercent > Props.phaseTwoHealthPercent)
                return;

            HediffDef phase = PhaseTwo;
            if (phase != null && !pawn.health.hediffSet.HasHediff(phase))
                pawn.health.AddHediff(phase);

            phaseTwoTriggered = true;

            if (pawn.Spawned)
                Messages.Message(pawn.LabelShortCap + " entered overdrive.", pawn, MessageTypeDefOf.ThreatBig, false);
        }
    }
}


===== FILE: Source/KrolikBosses/IncidentWorker_BossRaid.cs =====
using System.Collections.Generic;
using System.Linq;
using RimWorld;
using Verse;
using Verse.AI.Group;

namespace KrolikBosses
{
    public class IncidentWorker_BossRaid : IncidentWorker
    {
        protected override bool CanFireNowSub(IncidentParms parms)
        {
            if (!(parms.target is Map map))
                return false;

            if (Find.FactionManager.FirstFactionOfDef(DefDatabase<FactionDef>.GetNamed("KRL_BlackClan")) == null)
                return false;

            float points = StorytellerUtility.DefaultThreatPointsNow(map);
            return DefDatabase<BossDef>.AllDefsListForReading.Any(x => x.minimumThreatPoints <= points);
        }

        protected override bool TryExecuteWorker(IncidentParms parms)
        {
            if (!(parms.target is Map map))
                return false;

            FactionDef factionDef = DefDatabase<FactionDef>.GetNamed("KRL_BlackClan");
            Faction faction = Find.FactionManager.FirstFactionOfDef(factionDef);
            if (faction == null)
                return false;

            float points = parms.points > 0f ? parms.points : StorytellerUtility.DefaultThreatPointsNow(map);

            List<BossDef> eligible = DefDatabase<BossDef>.AllDefsListForReading
                .Where(x => x.pawnKind != null && x.minimumThreatPoints <= points)
                .ToList();

            if (eligible.Count == 0)
                return false;

            BossDef boss = eligible.RandomElementByWeight(x => x.selectionWeight > 0f ? x.selectionWeight : 0.01f);
            int count = boss.count < 1 ? 1 : boss.count;
            List<Pawn> spawned = new List<Pawn>();
            IntVec3 edge = CellFinder.RandomEdgeCell(map);

            for (int i = 0; i < count; i++)
            {
                Pawn pawn = PawnGenerator.GeneratePawn(boss.pawnKind, faction, map.Tile);

                if (boss.passiveHediff != null && pawn.health != null)
                    pawn.health.AddHediff(boss.passiveHediff);

                if (boss.forcedWeapon != null && pawn.equipment != null)
                {
                    pawn.equipment.DestroyAllEquipment();
                    Thing weapon = ThingMaker.MakeThing(boss.forcedWeapon);
                    pawn.equipment.AddEquipment((ThingWithComps)weapon);
                }

                IntVec3 cell = CellFinder.RandomClosewalkCellNear(edge, map, 8);
                GenSpawn.Spawn(pawn, cell, map, WipeMode.Vanish);
                spawned.Add(pawn);
            }

            if (spawned.Count == 0)
                return false;

            LordMaker.MakeNewLord(
                faction,
                new LordJob_AssaultColony(
                    faction,
                    canKidnap: false,
                    canTimeoutOrFlee: false,
                    sappers: false,
                    useAvoidGridSmart: true,
                    canSteal: false),
                map,
                spawned);

            string label = boss.letterLabel.NullOrEmpty() ? "Boss raid" : boss.letterLabel;
            string text = boss.letterText.NullOrEmpty()
                ? "A small group of extremely dangerous enemies has arrived."
                : boss.letterText;

            Find.LetterStack.ReceiveLetter(label, text, LetterDefOf.ThreatBig, spawned[0]);
            Find.TickManager.slower.SignalForceNormalSpeedShort();
            return true;
        }
    }
}
